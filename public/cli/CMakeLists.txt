cmake_minimum_required(VERSION 3.16)
project(nimbros_cli CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
add_executable(nimbros nimbros.cpp)
target_link_libraries(nimbros PRIVATE CURL::libcurl)
enable_testing()
add_test(NAME selftest COMMAND nimbros selftest)
